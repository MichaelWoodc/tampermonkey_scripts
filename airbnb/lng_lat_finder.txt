// ==UserScript==
// @name         Airbnb Lat/Lng + URL Copy + SAGIS Injector
// @namespace    http://tampermonkey.net/
// @version      4.0
// @description  Floating draggable window showing Airbnb listing coordinates, copy button, and SAGIS injection
// @match        https://www.airbnb.com/*
// @grant        none
// ==/UserScript==

(function() {
    'use strict';

    // Create floating window
    const box = document.createElement('div');
    box.id = "latLngBox";
    box.style.position = "fixed";
    box.style.top = "80px";
    box.style.right = "20px";
    box.style.zIndex = "999999";
    box.style.background = "rgba(0,0,0,0.85)";
    box.style.color = "white";
    box.style.padding = "12px 16px";
    box.style.borderRadius = "8px";
    box.style.fontSize = "14px";
    box.style.fontFamily = "Arial, sans-serif";
    box.style.cursor = "move";
    box.style.boxShadow = "0 0 10px rgba(0,0,0,0.5)";
    box.innerHTML = `
        Lat/Lng:<br>
        <b id="coordsText">Waiting for map…</b><br><br>

        <button id="copyBtn" style="
            padding:4px 8px;
            font-size:12px;
            cursor:pointer;
            border:none;
            border-radius:4px;
            margin-bottom:6px;
        ">Copy URL + Coords</button><br>

        <button id="sagisBtn" style="
            padding:4px 8px;
            font-size:12px;
            cursor:pointer;
            border:none;
            border-radius:4px;
        ">Open in SAGIS</button>
    `;
    document.body.appendChild(box);

    // Make draggable
    let isDragging = false;
    let offsetX, offsetY;

    box.addEventListener("mousedown", (e) => {
        isDragging = true;
        offsetX = e.clientX - box.offsetLeft;
        offsetY = e.clientY - box.offsetTop;
    });

    document.addEventListener("mousemove", (e) => {
        if (isDragging) {
            box.style.left = (e.clientX - offsetX) + "px";
            box.style.top = (e.clientY - offsetY) + "px";
        }
    });

    document.addEventListener("mouseup", () => {
        isDragging = false;
    });

    // Extract coordinates
    function getCoords() {
        const marker = document.querySelector('gmp-advanced-marker[position]');
        if (!marker) return null;

        const pos = marker.getAttribute('position');
        if (!pos) return null;

        const [lat, lng] = pos.split(',');
        return { lat, lng };
    }

    // Copy function
    function copyData() {
        const coords = document.getElementById("coordsText").innerText;
        const url = window.location.href;

        const [lat, lng] = coords.split(',').map(v => v.trim());

        const finalText = `${url}, ${lng}, ${lat}`;

        navigator.clipboard.writeText(finalText).then(() => {
            document.getElementById("copyBtn").innerText = "Copied!";
            setTimeout(() => {
                document.getElementById("copyBtn").innerText = "Copy URL + Coords";
            }, 1200);
        });
    }

    // SAGIS injection
    function openSagis() {
        const coords = document.getElementById("coordsText").innerText;
        const [lat, lng] = coords.split(',').map(v => v.trim());

        // SAGIS viewer base URL
        const sagisBase = "https://www.sagis.org/?page=SAGIS-Viewer#";

        // SAGIS expects: lon,lat (Web Mercator or WGS84 depending on layer)
        // For now we inject raw WGS84 into a custom param
        const sagisURL = `${sagisBase}lat=${lat}&lng=${lng}`;

        window.open(sagisURL, "_blank");
    }

    // Attach button handlers
    document.getElementById("copyBtn").addEventListener("click", copyData);
    document.getElementById("sagisBtn").addEventListener("click", openSagis);

    // Poll every second (safe)
    setInterval(() => {
        const coords = getCoords();
        if (coords) {
            document.getElementById("coordsText").innerText = `${coords.lat}, ${coords.lng}`;
        }
    }, 1000);

})();
