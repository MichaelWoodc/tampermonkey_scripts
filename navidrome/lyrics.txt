// ==UserScript==
// @name         Navidrome Lyric Accessibility Booster (v6 - Percent Width + Safe Drag)
// @namespace    http://tampermonkey.net/
// @version      6.0
// @description  Bottom-centered lyrics, percent width, safe dragging, draggable/minimizable panel, forced lyrics
// @match        https://yourdomain.com/*
// @grant        none
// ==/UserScript==

// --- CREATE CONTROL PANEL ---

(function() {
    'use strict';

    let settings = {
        fontSize: 32,
        wrapWidth: 80,          // percent width
        textColor: "#ff0000",
        bgColor: "transparent"
    };

    // --- CREATE CONTROL PANEL ---
    const panel = document.createElement("div");
    panel.style.cssText = `
        position: fixed;
        top: 20px;
        right: 20px;
        z-index: 99999;
        background: rgba(0,0,0,0.75);
        color: white;
        padding: 12px;
        border-radius: 8px;
        width: 240px;
        font-family: sans-serif;
        font-size: 14px;
        cursor: move;
    `;

    panel.innerHTML = `
        <div style="display:flex; justify-content:space-between; align-items:center;">
            <div style="font-weight:bold;">Lyric Controls</div>
            <button id="min-btn" style="background:none; border:1px solid #fff; color:white; padding:2px 6px; border-radius:4px; cursor:pointer;">–</button>
        </div>

        <div id="panel-body" style="margin-top:10px;">
            Font Size: <span id="fs-val">${settings.fontSize}px</span><br>
            <input id="fs" type="range" min="12" max="120" value="${settings.fontSize}" style="width:100%;">

            <br><br>

            Wrap Width: <span id="ww-val">${settings.wrapWidth}%</span><br>
            <input id="ww" type="range" min="20" max="100" value="${settings.wrapWidth}" style="width:100%;">

            <br><br>

            Text Color:<br>
            <input id="tc" type="color" value="${settings.textColor}" style="width:100%;">

            <br><br>

            Background Color:<br>
            <input id="bc" type="color" value="#00000000" style="width:100%;">
        </div>
    `;

    document.body.appendChild(panel);

    // --- DRAGGABLE PANEL ---
    let isDraggingPanel = false, offsetX = 0, offsetY = 0;

    panel.addEventListener("mousedown", e => {
        if (e.target.tagName === "INPUT" || e.target.tagName === "BUTTON") return;
        isDraggingPanel = true;
        offsetX = e.clientX - panel.offsetLeft;
        offsetY = e.clientY - panel.offsetTop;
    });

    document.addEventListener("mousemove", e => {
        if (!isDraggingPanel) return;
        panel.style.left = (e.clientX - offsetX) + "px";
        panel.style.top = (e.clientY - offsetY) + "px";
        panel.style.right = "auto";
    });

    document.addEventListener("mouseup", () => {
        isDraggingPanel = false;
    });

    // --- MINIMIZE BUTTON ---
    document.getElementById("min-btn").addEventListener("click", () => {
        const body = document.getElementById("panel-body");
        body.style.display = body.style.display === "none" ? "block" : "none";
    });

    // --- APPLY STYLES TO LYRICS ---
    function applyStyles() {
        const lyric = document.querySelector(".music-player-lyric.react-draggable");
        if (!lyric) return;

        const currentTransform = lyric.style.transform;

        // Accessibility styling
        lyric.style.fontSize = settings.fontSize + "px";
        lyric.style.display = "inline-block";     // shrink hitbox to text
        lyric.style.width = "auto";
        lyric.style.maxWidth = settings.wrapWidth + "%";
        lyric.style.color = settings.textColor;
        lyric.style.background = settings.bgColor;
        lyric.style.padding = "10px";
        lyric.style.borderRadius = "6px";
        lyric.style.whiteSpace = "normal";
        lyric.style.lineHeight = "1.4";
        lyric.style.textAlign = "center";

        // Detect if user dragged it
        const userMoved = currentTransform && currentTransform !== "translate(0px, 0px)";

        if (!userMoved) {
            // Default: bottom center like subtitles
            lyric.style.position = "fixed";
            lyric.style.left = "50%";
            lyric.style.bottom = "20px";
            lyric.style.top = "auto";
            lyric.style.transform = "translateX(-50%)";
        } else {
            lyric.style.transform = currentTransform;
        }
    }

    // --- FORCE LYRICS ALWAYS ON ---
    function ensureLyricsOn() {
        const btn = document.querySelector(".lyric-btn");
        if (!btn) return;

        const isActive = btn.classList.contains("lyric-btn-active");
        if (!isActive) btn.click();
    }

    // --- SAFE DRAGGING FOR LYRICS ---
    let draggingLyrics = false;
    let dragOffsetX = 0;
    let dragOffsetY = 0;

    document.addEventListener("mousedown", e => {
        const lyric = document.querySelector(".music-player-lyric.react-draggable");
        if (!lyric) return;

        // Only drag when clicking the actual text
        if (e.target === lyric || lyric.contains(e.target)) {
            draggingLyrics = true;
            dragOffsetX = e.clientX - lyric.offsetLeft;
            dragOffsetY = e.clientY - lyric.offsetTop;
        }
    });

    document.addEventListener("mousemove", e => {
        if (!draggingLyrics) return;

        const lyric = document.querySelector(".music-player-lyric.react-draggable");
        if (!lyric) return;

        lyric.style.position = "fixed";
        lyric.style.left = (e.clientX - dragOffsetX) + "px";
        lyric.style.top = (e.clientY - dragOffsetY) + "px";
        lyric.style.bottom = "auto";
        lyric.style.transform = "none";
    });

    document.addEventListener("mouseup", () => {
        draggingLyrics = false;
    });

    // --- OBSERVE DOM CHANGES ---
    const observer = new MutationObserver(() => {
        ensureLyricsOn();
        applyStyles();
    });

    observer.observe(document.body, { childList: true, subtree: true });

    // --- EVENT HANDLERS ---
    document.getElementById("fs").addEventListener("input", e => {
        settings.fontSize = e.target.value;
        document.getElementById("fs-val").textContent = settings.fontSize + "px";
        applyStyles();
    });

    document.getElementById("ww").addEventListener("input", e => {
        settings.wrapWidth = e.target.value;
        document.getElementById("ww-val").textContent = settings.wrapWidth + "%";
        applyStyles();
    });

    document.getElementById("tc").addEventListener("input", e => {
        settings.textColor = e.target.value;
        applyStyles();
    });

    document.getElementById("bc").addEventListener("input", e => {
        settings.bgColor = e.target.value + "80";
        applyStyles();
    });

    // Initial run
    ensureLyricsOn();
    applyStyles();
})();
