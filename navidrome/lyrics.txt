// ==UserScript==
// @name         Navidrome Lyric Accessibility Booster
// @namespace    http://tampermonkey.net/
// @version      2
// @description  Bottom-centered lyrics, viewport-based wrap width, tight drag hitbox, draggable/minimizable panel, forced lyrics, no clipping, force-center toggle
// @match        https://navidrome.michaelwoodcock.duckdns.org/*
// @grant        none
// ==/UserScript==

(function() {
    'use strict';

    let settings = {
        fontSize: 32,
        wrapWidth: 90,          // percent of viewport width (vw)
        textColor: "#ff0000",
        bgColor: "transparent",
        forceCenter: true       // default ON
    };

    // --- CREATE CONTROL PANEL ---
    const panel = document.createElement("div");
    panel.style.cssText = `
        position: fixed;
        top: 20px;
        right: 20px;
        z-index: 99999;
        background: rgba(0,0,0,0.75);
        color: white;
        padding: 12px;
        border-radius: 8px;
        width: 260px;
        font-family: sans-serif;
        font-size: 14px;
        cursor: move;
    `;

    panel.innerHTML = `
        <div style="display:flex; justify-content:space-between; align-items:center;">
            <div style="font-weight:bold;">Lyric Controls</div>
            <button id="min-btn" style="background:none; border:1px solid #fff; color:white; padding:2px 6px; border-radius:4px; cursor:pointer;">–</button>
        </div>

        <div id="panel-body" style="margin-top:10px;">

            <label style="display:flex; align-items:center; gap:6px; margin-bottom:10px;">
                <input id="force-center" type="checkbox" checked>
                Force Center
            </label>

            Font Size: <span id="fs-val">${settings.fontSize}px</span><br>
            <input id="fs" type="range" min="12" max="120" value="${settings.fontSize}" style="width:100%;">

            <br><br>

            Wrap Width: <span id="ww-val">${settings.wrapWidth}% of screen</span><br>
            <input id="ww" type="range" min="20" max="100" value="${settings.wrapWidth}" style="width:100%;">

            <br><br>

            Text Color:<br>
            <input id="tc" type="color" value="${settings.textColor}" style="width:100%;">

            <br><br>

            Background Color:<br>
            <input id="bc" type="color" value="#00000000" style="width:100%;">
        </div>
    `;

    document.body.appendChild(panel);

    // --- DRAGGABLE PANEL ---
    let isDraggingPanel = false, offsetX = 0, offsetY = 0;

    panel.addEventListener("mousedown", e => {
        if (["INPUT", "BUTTON"].includes(e.target.tagName) || e.target.type === "checkbox") return;
        isDraggingPanel = true;
        offsetX = e.clientX - panel.offsetLeft;
        offsetY = e.clientY - panel.offsetTop;
    });

    document.addEventListener("mousemove", e => {
        if (!isDraggingPanel) return;
        panel.style.left = (e.clientX - offsetX) + "px";
        panel.style.top = (e.clientY - offsetY) + "px";
        panel.style.right = "auto";
    });

    document.addEventListener("mouseup", () => {
        isDraggingPanel = false;
    });

    // --- MINIMIZE BUTTON ---
    document.getElementById("min-btn").addEventListener("click", () => {
        const body = document.getElementById("panel-body");
        body.style.display = body.style.display === "none" ? "block" : "none";
    });

    // --- APPLY STYLES TO LYRICS ---
    function applyStyles() {
        const lyric = document.querySelector(".music-player-lyric.react-draggable");
        if (!lyric) return;

        // Create wrapper if missing
        if (!lyric.querySelector(".inner-lyric-text")) {
            const inner = document.createElement("div");
            inner.className = "inner-lyric-text";

            while (lyric.firstChild) {
                inner.appendChild(lyric.firstChild);
            }

            lyric.appendChild(inner);
        }

        const inner = lyric.querySelector(".inner-lyric-text");

        // OUTER BOX (full wrap width)
        lyric.style.position = "fixed";
        lyric.style.boxSizing = "border-box";
        lyric.style.width = settings.wrapWidth + "vw";
        lyric.style.maxWidth = settings.wrapWidth + "vw";
        lyric.style.whiteSpace = "normal";
        lyric.style.overflow = "visible";
        lyric.style.height = "auto";

        // ⭐ CRITICAL: Outer box should NOT receive clicks
        lyric.style.pointerEvents = "none";

        // INNER BOX (tight to text)
        inner.style.display = "inline-block";
        inner.style.width = "auto";
        inner.style.maxWidth = "100%";
        inner.style.whiteSpace = "normal";
        inner.style.textAlign = "center";
        inner.style.fontSize = settings.fontSize + "px";
        inner.style.color = settings.textColor;
        inner.style.background = settings.bgColor;
        inner.style.padding = "10px";
        inner.style.borderRadius = "6px";
        inner.style.lineHeight = "1.4";

        // ⭐ CRITICAL: Inner box receives clicks
        inner.style.pointerEvents = "auto";

        // FORCE CENTER
        if (settings.forceCenter) {
            lyric.style.left = "50%";
            lyric.style.transform = "translateX(-50%)";
            lyric.style.bottom = "20px";
            lyric.style.top = "auto";
        }
    }

    // --- FORCE LYRICS ALWAYS ON ---
    function ensureLyricsOn() {
        const btn = document.querySelector(".lyric-btn");
        if (!btn) return;

        const isActive = btn.classList.contains("lyric-btn-active");
        if (!isActive) btn.click();
    }

    // --- SAFE DRAGGING FOR INNER TEXT ONLY ---
    let draggingLyrics = false;
    let dragOffsetX = 0;
    let dragOffsetY = 0;

    document.addEventListener("mousedown", e => {
        const inner = document.querySelector(".inner-lyric-text");
        const lyric = document.querySelector(".music-player-lyric.react-draggable");
        if (!inner || !lyric) return;

        // Only drag when clicking the inner text
        if (e.target === inner || inner.contains(e.target)) {
            draggingLyrics = true;
            dragOffsetX = e.clientX - lyric.offsetLeft;
            dragOffsetY = e.clientY - lyric.offsetTop;
        }
    });

    document.addEventListener("mousemove", e => {
        if (!draggingLyrics) return;

        const lyric = document.querySelector(".music-player-lyric.react-draggable");
        if (!lyric) return;

        lyric.style.left = (e.clientX - dragOffsetX) + "px";
        lyric.style.top = (e.clientY - dragOffsetY) + "px";
        lyric.style.bottom = "auto";
        lyric.style.transform = "none";
    });

    document.addEventListener("mouseup", () => {
        draggingLyrics = false;
    });

    // --- OBSERVE DOM CHANGES ---
    const observer = new MutationObserver(() => {
        ensureLyricsOn();
        applyStyles();
    });

    observer.observe(document.body, { childList: true, subtree: true });

    // --- EVENT HANDLERS ---
    document.getElementById("force-center").addEventListener("change", e => {
        settings.forceCenter = e.target.checked;
        applyStyles();
    });

    document.getElementById("fs").addEventListener("input", e => {
        settings.fontSize = e.target.value;
        document.getElementById("fs-val").textContent = settings.fontSize + "px";
        applyStyles();
    });

    document.getElementById("ww").addEventListener("input", e => {
        settings.wrapWidth = e.target.value;
        document.getElementById("ww-val").textContent = settings.wrapWidth + "% of screen";
        applyStyles();
    });

    document.getElementById("tc").addEventListener("input", e => {
        settings.textColor = e.target.value;
        applyStyles();
    });

    document.getElementById("bc").addEventListener("input", e => {
        settings.bgColor = e.target.value + "80";
        applyStyles();
    });

    // Initial run
    ensureLyricsOn();
    applyStyles();
})();
