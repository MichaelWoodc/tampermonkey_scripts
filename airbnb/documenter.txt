// ==UserScript==
// @name         Airbnb - Combined Full Suite (Desc + Sleep + Clean Export)
// @namespace    http://tampermonkey.net/
// @version      4.0
// @description  Injects full description, renders sleeping arrangements, and exports reviews cleanly.
// @match        https://www.airbnb.com/rooms/*
// @grant        none
// ==/UserScript==

(function() {
    'use strict';

    // ----------------- PART 1: DESCRIPTION INJECTOR -----------------
    const modalObserver = new MutationObserver((mutations) => {
        const modal = document.querySelector("div[role='dialog']");
        if (modal) {
            const descSection = document.querySelector("div[data-section-id='DESCRIPTION_DEFAULT']");
            const isDescModal = modal.innerText.includes("About this space");

            if (descSection && isDescModal && !descSection.querySelector(".injected-full-description")) {
                const fullHTML = modal.innerHTML;
                const closeBtn = modal.querySelector("button[aria-label='Close']");
                if (closeBtn) closeBtn.click();

                const originalText = descSection.querySelector("div[style*='-webkit-line-clamp']");
                if (originalText) originalText.style.display = "none";

                const displayDiv = document.createElement("div");
                displayDiv.className = "injected-full-description";
                displayDiv.innerHTML = fullHTML;
                displayDiv.style.margin = "20px 0";
                descSection.appendChild(displayDiv);
            }
        }
    });
    modalObserver.observe(document.body, { childList: true, subtree: true });

    // ----------------- PART 2: EXPANDED SLEEPING ARRANGEMENTS -----------------
    function renderSleepingArrangementsExpanded() {
        // Find the main container of the carousel
        const carouselList = document.querySelector("ul.omd1zsm");
        if (!carouselList || document.querySelector(".injected-sleep-grid")) return;

        // Create a new grid container to hold all items
        const gridContainer = document.createElement("div");
        gridContainer.className = "injected-sleep-grid";
        gridContainer.style.cssText = "display: grid; grid-template-columns: repeat(auto-fill, minmax(300px, 1fr)); gap: 20px; margin-top: 20px;";

        // Move all <li> items from the carousel into our new grid
        const items = Array.from(carouselList.querySelectorAll("li"));
        items.forEach(item => {
            // Remove the fixed width styles that keep them in the carousel
            item.style.flex = "none";
            item.style.maxWidth = "100%";
            gridContainer.appendChild(item);
        });

        // Hide the original carousel wrapper and navigation buttons
        const parentSection = document.querySelector("section[aria-roledescription='carousel']");
        if (parentSection) {
            parentSection.innerHTML = ""; // Clear the carousel
            parentSection.appendChild(gridContainer); // Add our grid instead
        }
    }

    // Run this when the page loads to capture the elements
    window.addEventListener("load", () => {
        // A small delay to ensure the React-rendered elements exist
        setTimeout(renderSleepingArrangementsExpanded, 2000);
    });

    // ----------------- PART 3: SURGICAL REVIEW EXPORT -----------------
    setInterval(() => {
        if (document.getElementById("airbnb-review-export-btn")) return;
        const panel = document.querySelector("div[data-testid='pdp-reviews-modal-scrollable-panel']");
        if (!panel) return;

        const btn = document.createElement("button");
        btn.id = "airbnb-review-export-btn";
        btn.textContent = "Export Reviews (Expanded)";
        btn.style.cssText = "position:fixed; top:80px; right:20px; z-index:999999; padding:10px 14px; background:#ff385c; color:white; border-radius:6px; cursor:pointer; font-weight:bold;";

        btn.onclick = () => {
            panel.querySelectorAll("button").forEach(b => { if (b.innerText.trim().toLowerCase() === "show more") b.click(); });

            setTimeout(() => {
                const reviews = Array.from(panel.querySelectorAll('[data-review-id]'));
                const popup = window.open("", "_blank");

                const cleanContent = (html) => {
                    let cleaned = html.replace(/<button[\s\S]*?<\/button>/g, '').replace(/·/g, '').replace(/\d+\s*(weeks|months|years|days)?\s*ago/gi, '');
                    let tempDiv = document.createElement('div');
                    tempDiv.innerHTML = cleaned;
                    // Remove empty nodes
                    const walk = document.createTreeWalker(tempDiv, NodeFilter.SHOW_TEXT, null, false);
                    let node, toRemove = [];
                    while (node = walk.nextNode()) if (node.nodeValue.replace(/,/g, '').trim() === '') toRemove.push(node);
                    toRemove.forEach(n => n.remove());
                    return tempDiv.innerHTML.replace(/(<br\s*\/?>\s*){2,}/gi, '<br>');
                };

                popup.document.write(`<html><body><h2>Airbnb Reviews</h2>${reviews.map(r => `<div style="margin-bottom:20px; border-bottom:1px solid #ccc;">${cleanContent(r.innerHTML)}</div>`).join('')}<script>setTimeout(window.print, 1000);</script></body></html>`);
                popup.document.close();
            }, 1500);
        };
        document.body.appendChild(btn);
    }, 3000);
})();
