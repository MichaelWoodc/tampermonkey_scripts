// ==UserScript==
// @name         Google Maps Review Exporter (v2.0 – Stars + DOM Clone + Expand More)
// @namespace    http://tampermonkey.net/
// @version      2.0
// @description  Export Google Maps reviews with real stars, images, formatting, and expanded text
// @match        https://www.google.com/maps/*
// @grant        none
// ==/UserScript==

(function() {
    'use strict';

    let stopFlag = false;

    function sleep(ms) {
        return new Promise(resolve => setTimeout(resolve, ms));
    }

    function addStopButton() {
        if (document.getElementById("gm-stop-btn")) return;

        const stop = document.createElement("button");
        stop.id = "gm-stop-btn";
        stop.textContent = "STOP";
        stop.style.position = "fixed";
        stop.style.top = "70px";
        stop.style.right = "20px";
        stop.style.zIndex = "999999";
        stop.style.padding = "10px 16px";
        stop.style.background = "#d93025";
        stop.style.color = "#fff";
        stop.style.border = "none";
        stop.style.borderRadius = "6px";
        stop.style.fontSize = "14px";
        stop.style.cursor = "pointer";
        stop.style.display = "none";

        stop.onclick = () => {
            stopFlag = true;
            stop.style.display = "none";
            alert("Stopped.");
        };

        document.body.appendChild(stop);
    }

    function showStop() {
        const stop = document.getElementById("gm-stop-btn");
        if (stop) stop.style.display = "block";
    }

    function hideStop() {
        const stop = document.getElementById("gm-stop-btn");
        if (stop) stop.style.display = "none";
    }

    function detectReviewPanel() {
        return document.querySelector(
            "div.m6QErb[aria-label*='Reviews'], div.m6QErb.tLjsW"
        );
    }

    //  Replace Google’s private star glyphs with real Unicode stars
    function replaceStarsInBlock(block) {
        const starContainer = block.querySelector("span[aria-label*='star']");
        if (!starContainer) return;

        const label = starContainer.getAttribute("aria-label"); // e.g. "5 stars"
        const match = label.match(/(\d+(\.\d+)?)/);
        if (!match) return;

        const rating = parseFloat(match[1]);

        const full = Math.floor(rating);
        const half = rating % 1 >= 0.5 ? 1 : 0;
        const empty = 5 - full - half;

        const stars =
            "★".repeat(full) +
            (half ? "½" : "") +
            "☆".repeat(empty);

        starContainer.innerHTML = `<span style="font-size:16px;">${stars}</span>`;
    }

    //  Auto-expand ALL "More" buttons
    async function expandAllMoreButtons() {
        let buttons = Array.from(document.querySelectorAll("button[aria-label='See more'], button.w8nwRe"));

        while (buttons.length > 0 && !stopFlag) {
            for (const btn of buttons) {
                btn.dispatchEvent(new MouseEvent("mousedown", { bubbles: true }));
                btn.dispatchEvent(new MouseEvent("mouseup", { bubbles: true }));
                btn.dispatchEvent(new MouseEvent("click", { bubbles: true }));
                await sleep(200);
            }

            await sleep(500);

            buttons = Array.from(document.querySelectorAll("button[aria-label='See more'], button.w8nwRe"));
        }
    }

    async function extractReviewNodes() {
        const blocks = document.querySelectorAll(".jftiEf");
        const clones = [];

        for (const block of blocks) {
            replaceStarsInBlock(block); //  Fix stars BEFORE cloning

            const clone = block.cloneNode(true);

            clone.querySelectorAll("img").forEach(img => {
                if (img.dataset.src) img.src = img.dataset.src;
                if (img.getAttribute("src") === "") img.src = img.dataset.src || "";
            });

            clones.push(clone);
        }

        return clones;
    }

    async function openPopupWithNodes(nodes) {
        const pageTitle = document.title.replace(" – Google Maps", "").trim();
        const exportTitle = pageTitle + " – Reviews Export";

        const win = window.open("", "_blank");

        win.document.write(`
            <html>
            <head>
                <title>${exportTitle}</title>
                <style>
                    body { font-family: Arial, sans-serif; padding: 20px; }
                    .review { border-bottom: 1px solid #ccc; padding: 15px 0; }
                    img { max-width: 100%; }
                </style>
            </head>
            <body>
                <h1>${exportTitle}</h1>
                <div id="reviews"></div>
            </body>
            </html>
        `);

        win.document.close();

        const container = win.document.getElementById("reviews");

        nodes.forEach(node => {
            const wrapper = win.document.createElement("div");
            wrapper.className = "review";
            wrapper.appendChild(node);
            container.appendChild(wrapper);
        });

        setTimeout(() => win.print(), 800);
    }

    function addExportButton() {
        if (document.getElementById("gm-export-btn")) return;

        const btn = document.createElement("button");
        btn.id = "gm-export-btn";
        btn.textContent = "Export Reviews";
        btn.style.position = "fixed";
        btn.style.top = "20px";
        btn.style.right = "20px";
        btn.style.zIndex = "999999";
        btn.style.padding = "12px 18px";
        btn.style.background = "#1a73e8";
        btn.style.color = "#fff";
        btn.style.border = "none";
        btn.style.borderRadius = "6px";
        btn.style.fontSize = "16px";
        btn.style.cursor = "pointer";
        btn.style.boxShadow = "0 2px 6px rgba(0,0,0,0.3)";

        btn.onclick = async () => {
            stopFlag = false;
            showStop();

            const panel = detectReviewPanel();

            if (!panel) {
                hideStop();
                alert("Click the Reviews tab first, then scroll ALL the way down.");
                return;
            }

            if (panel.scrollTop + panel.clientHeight + 50 < panel.scrollHeight) {
                hideStop();
                alert("Scroll ALL the way down until no more reviews load, then click Export again.");
                return;
            }

            //  Expand all "More" buttons BEFORE exporting
            await expandAllMoreButtons();

            const nodes = await extractReviewNodes();
            hideStop();

            if (nodes.length === 0) {
                alert("No reviews found.");
                return;
            }

            openPopupWithNodes(nodes);
        };

        document.body.appendChild(btn);
    }

    const observer = new MutationObserver(() => {
        const contribTab = document.querySelector("div.Gpq6kf.NlVald");
        const placeBtn = document.querySelector("button[jsaction*='pane.rating.moreReviews']");

        if (contribTab || placeBtn) {
            addExportButton();
            addStopButton();
        }
    });

    observer.observe(document.body, { childList: true, subtree: true });

})();
